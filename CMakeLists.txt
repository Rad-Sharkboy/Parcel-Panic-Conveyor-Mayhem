cmake_minimum_required(VERSION 3.22)
project(ParcelPanic VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PARCEL_BUILD_GAME "Build the Raylib graphical game" ON)
option(PARCEL_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
add_library(parcel_core
    src/module1_conveyor/CircularQueue.cpp src/module1_conveyor/CommandStack.cpp
    src/module1_conveyor/ConveyorSystem.cpp src/module2_dispatch/MinHeap.cpp
    src/module2_dispatch/HashTable.cpp src/module2_dispatch/DispatchSystem.cpp
    src/shared/Level.cpp src/shared/Simulation.cpp src/shared/Demos.cpp)
target_include_directories(parcel_core PUBLIC include)
function(parcel_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        if(PARCEL_SANITIZERS)
            target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PUBLIC -fsanitize=address,undefined)
        endif()
    endif()
endfunction()
parcel_warnings(parcel_core)
if(PARCEL_BUILD_GAME)
    # Raylib is downloaded automatically on the first build, just like the OOP project.
    include(FetchContent)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
    if(POLICY CMP0135)
        cmake_policy(SET CMP0135 NEW)
    endif()
    FetchContent_Declare(raylib
        URL https://codeload.github.com/raysan5/raylib/tar.gz/c1ab645ca298a2801097931d1079b10ff7eb9df8
        URL_HASH SHA256=be3443c3fb97f7cbd18ae3f0ea2a73e6c1486128df33b639dce7e44070fc285d)
    FetchContent_MakeAvailable(raylib)
    add_executable(parcel_panic src/main.cpp src/shared/Game.cpp src/shared/Renderer.cpp)
    target_link_libraries(parcel_panic PRIVATE parcel_core raylib)
    parcel_warnings(parcel_panic)
    if(WIN32)
        set_target_properties(parcel_panic PROPERTIES OUTPUT_NAME ParcelPanic)
    endif()
    # Bundle the MinGW runtimes so the finished Windows EXE can be shared on its own.
    if(MINGW)
        target_link_options(parcel_panic PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
endif()
